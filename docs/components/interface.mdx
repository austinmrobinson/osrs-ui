<Component path="interface/demo" />

## Installation

```bash
npx shadcn@latest add {{registry}}/interface.json
```

## Usage

```tsx
import { Interface, InterfaceHeader, InterfaceTitle, InterfaceClose, InterfaceContent } from "@/components/ui/interface"

<Interface>
  <InterfaceHeader>
    <InterfaceTitle>The Bank of Gielinor</InterfaceTitle>
    <InterfaceClose onClick={close} />
  </InterfaceHeader>
  <InterfaceContent>…</InterfaceContent>
</Interface>
```
