## Installation

```bash
npx shadcn@latest add {{registry}}/nav-list.json
```

## Usage

```tsx
<Tabs orientation="vertical" defaultValue="audio">
  <NavList>
    <NavItem value="audio">Audio</NavItem>
  </NavList>
  <TabsContent value="audio">…</TabsContent>
</Tabs>
```
