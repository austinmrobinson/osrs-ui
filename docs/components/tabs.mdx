<Component path="tabs/playground" />

## Installation

```bash
npx shadcn@latest add {{registry}}/tabs.json
```

## Usage

```tsx
import { Tabs, TabsList, TabsTrigger, TabsContent } from "@/components/ui/tabs"

<Tabs defaultValue="bosses">
  <TabsList>
    <TabsTrigger value="bosses">Bosses</TabsTrigger>
    <TabsTrigger value="raids" size="large">Raids</TabsTrigger>
  </TabsList>
  <TabsContent value="bosses">…</TabsContent>
</Tabs>
```
