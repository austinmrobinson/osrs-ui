## Installation

```bash
npx shadcn@latest add {{registry}}/select.json
```

## Usage

```tsx
<Select defaultValue="exact" items={{ exact: "Exact Value" }}>
  <SelectTrigger />
  <SelectContent>
    <SelectItem value="exact">Exact Value</SelectItem>
  </SelectContent>
</Select>
```
